        title: "Scorpion Solitaire",
        category: "Board",
        emoji: "🃏",
        grad: "linear-gradient(135deg, #0369a1, #050811)",
        desc: "Challenging card-stacking solitaire with smooth drag-and-drop touch play.",
        url: "https://playpager.com/embed/solitaire/index.html"
      }
    ];

    let activeCategory = "All";
    let currentGameUrl = "";

    function setCategory(cat, btnEl) {
      activeCategory = cat;
      document.querySelectorAll(".chip").forEach(c => c.classList.remove("active"));
      btnEl.classList.add("active");
      renderGames();
    }

    function resetFilters() {
      activeCategory = "All";
      document.getElementById("searchInput").value = "";
      document.querySelectorAll(".chip").forEach((c, i) => c.classList.toggle("active", i === 0));
      renderGames();
    }

    function renderGames() {
      const query = document.getElementById("searchInput").value.trim().toLowerCase();
      const grid = document.getElementById("gameGrid");
      const filtered = GAMES.filter(g => {
        const matchesCat = activeCategory === "All" || g.category === activeCategory;
        const matchesText = g.title.toLowerCase().includes(query) || g.desc.toLowerCase().includes(query) || g.category.toLowerCase().includes(query);
        return matchesCat && matchesText;
      });

      document.getElementById("gameCountText").innerHTML = `Showing <strong>${filtered.length}</strong> live embedded HTML5 games`;

      grid.innerHTML = filtered.map((game, idx) => `
        <article class="game-card" onclick="launchGame(${GAMES.indexOf(game)})">
          <div class="card-banner" style="background: ${game.grad};">
            <span class="badge-tag">${game.category}</span>
            <span class="badge-ipad">iPad Ready</span>
            <div class="card-emoji">${game.emoji}</div>
          </div>
          <div class="card-body">
            <div>
              <h3 class="card-title">${game.title}</h3>
              <p class="card-desc">${game.desc}</p>
            </div>
            <button class="play-btn" type="button">▶ Play Now</button>
          </div>
        </article>
      `).join("");
    }

    function launchGame(index) {
      const game = GAMES[index];
      if (!game) return;
      currentGameUrl = game.url;
      document.getElementById("playerEmoji").textContent = game.emoji;
      document.getElementById("playerTitle").textContent = game.title;
      document.getElementById("openNewTabBtn").href = game.url;
      document.getElementById("gameFrame").src = game.url;
      document.getElementById("playerOverlay").classList.add("open");
      document.body.style.overflow = "hidden";
    }

    function reloadCurrentGame() {
      const frame = document.getElementById("gameFrame");
      frame.src = "about:blank";
      setTimeout(() => { frame.src = currentGameUrl; }, 60);
    }

    function closeGame() {
      const frame = document.getElementById("gameFrame");
      frame.src = "about:blank";
      document.getElementById("playerOverlay").classList.remove("open");
      document.body.style.overflow = "";
    }

    function toggleIpadModal(show) {
      document.getElementById("ipadModal").classList.toggle("open", show);
    }

    renderGames();
  </script>
